  ```ts
  export function filterMenuForUser(
      items: MenuItem[],
      userPermissions: string[] | undefined
  ): MenuItem[] {
      const permissions = userPermissions ?? [];

      return items.filter((item) => {
          if (item.requiredPermissions && item.requiredPermissions.length > 0) {
              const hasAll = item.requiredPermissions.every((p) =>
                  permissions.includes(p)
              );
              if (!hasAll) {
                  return false;
              }
          }

          if (item.anyPermissions && item.anyPermissions.length > 0) {
              const hasAny = item.anyPermissions.some((p) =>
                  permissions.includes(p)
              );
              if (!hasAny) {
                  return false;
              }
          }

          return true;
      });
  }
  ```

  private function dashboardRoute(Request $request): string
  {
      $user = $request->user();

      if (! $user) {
          return route('home');
      }

      if ($user->hasAnyRole(['admin', 'editor', 'writer'])) {
          return '/admin';
      }

      if ($user->hasRole('subscriber')) {
          return route('dashboard.subscriber');
      }

      return route('home');
  }
